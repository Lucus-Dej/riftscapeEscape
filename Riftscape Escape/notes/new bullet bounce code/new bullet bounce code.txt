var velox = lengthdir_x(speed, direction);
var veloy = lengthdir_y(speed, direction);

var normx = 0;
var normy = 0;

if (velox > 0 && x <= other.bbox_left) {
    normx = -1;
}
else if (velox < 0 && x >= other.bbox_right) {
    normx = 1;
}
else if (veloy > 0 && y <= other.bbox_top) {
    normy = -1;
}
else if (veloy < 0 && y >= other.bbox_bottom) {
    normy = 1;
}

var dot = velox * normx + veloy * normy;

var rx = velox - 2 * dot * normx;
var ry = veloy - 2 * dot * normy;

direction = point_direction(0, 0, rx, ry);
